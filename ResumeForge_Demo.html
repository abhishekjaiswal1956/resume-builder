class ResumeManager:
    """OOP-based resume management system"""

    def __init__(self):
        self.personal       = {}          # Dictionary — stores key-value pairs
        self.education      = []          # List — dynamic academic records
        self.skills         = []          # List — deduplicated skill set
        self.experience     = []          # List — work history
        self.projects       = []          # List — project portfolio
        self.certifications = []          # List — credential storage

    def add_education(self, edu: dict):
        self.education.append(edu)           # List.append() — O(1)

    def add_skill(self, skill: str):
        if skill not in self.skills:        # Deduplication check
            self.skills.append(skill)

    def generate_text_resume(self) -> str:
        """File Handling — Exception Handling"""
        lines = []
        try:
            with open("resume.txt", "w") as f:
                f.write("\n".join(lines))   # File Handling ✓
        except IOError as e:
            print(f"Error: {e}")            # Exception Handling ✓
        return "\n".join(lines)

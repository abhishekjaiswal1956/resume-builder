class ResumeManager:
    """OOP-based resume management system"""

    def __init__(self):
        self.personal      = {}
        self.education     = []
        self.skills        = []
        self.experience    = []
        self.projects      = []
        self.certifications = []

    def add_education(self, edu):
        self.education.append(edu)      # List management

    def add_skill(self, skill):
        if skill not in self.skills:
            self.skills.append(skill)   # Deduplication

    def generate_text_resume(self):
        """Generate formatted text file — File Handling"""
        lines = []
        with open("resume.txt", "w") as f:
            f.write("\n".join(lines))  # Exception Handling ✓
        return "\n".join(lines)
